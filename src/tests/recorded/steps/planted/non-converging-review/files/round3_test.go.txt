package r2

import (
	"encoding/json"
	"net/http"
	"net/http/httptest"
	"strings"
	"testing"

	"example.com/service/gen"
	"example.com/service/internal/app"
)

// R-2: creating a note returns that note, with its text as given.
func TestR2CreateReturnsTheNote(t *testing.T) {
	srv := httptest.NewServer(gen.Handler(app.New()))
	defer srv.Close()
	resp, err := http.Post(srv.URL+"/notes", "application/json", strings.NewReader(`{"text":"Buy Milk"}`))
	if err != nil {
		t.Fatal(err)
	}
	defer resp.Body.Close()
	var n struct {
		ID   int
		Text string
	}
	if err := json.NewDecoder(resp.Body).Decode(&n); err != nil {
		t.Fatal(err)
	}
	if n.ID != 1 || n.Text != "Buy Milk" {
		t.Errorf("the created note is %+v, want id 1 and the text \"Buy Milk\"", n)
	}
}
