package r3

import (
	"net/http"
	"net/http/httptest"
	"strings"
	"testing"

	"example.com/service/gen"
	"example.com/service/internal/app"
)

// R-3: a text of only spaces is empty (the slice's answer to Q-1), so the note is refused.
func TestR3SpacesAreRefused(t *testing.T) {
	srv := httptest.NewServer(gen.Handler(app.New()))
	defer srv.Close()
	resp, err := http.Post(srv.URL+"/notes", "application/json", strings.NewReader(`{"text":"   "}`))
	if err != nil {
		t.Fatal(err)
	}
	resp.Body.Close()
	if resp.StatusCode != http.StatusBadRequest {
		t.Errorf("a note of only spaces answered %d, want 400", resp.StatusCode)
	}
}
