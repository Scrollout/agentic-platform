package r1

import (
	"encoding/json"
	"net/http"
	"net/http/httptest"
	"strings"
	"testing"

	"example.com/service/gen"
	"example.com/service/internal/app"
)

// R-1: listing returns every note, however many were created.
func TestR1ListReturnsEveryNote(t *testing.T) {
	srv := httptest.NewServer(gen.Handler(app.New()))
	defer srv.Close()
	for _, text := range []string{"a", "b", "c"} {
		resp, err := http.Post(srv.URL+"/notes", "application/json", strings.NewReader(`{"text":"`+text+`"}`))
		if err != nil {
			t.Fatal(err)
		}
		resp.Body.Close()
	}
	resp, err := http.Get(srv.URL + "/notes")
	if err != nil {
		t.Fatal(err)
	}
	defer resp.Body.Close()
	var notes []struct{ Text string }
	if err := json.NewDecoder(resp.Body).Decode(&notes); err != nil {
		t.Fatal(err)
	}
	if len(notes) != 3 {
		t.Errorf("the list holds %d notes after three were created, want 3", len(notes))
	}
}
