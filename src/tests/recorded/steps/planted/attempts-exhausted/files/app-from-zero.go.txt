// Package app is the in-memory notes service: the code step replaces the seam New.
package app

import (
	"encoding/json"
	"net/http"
	"strings"
	"sync"

	"example.com/service/gen"
)

// Note is one note: a text with an id.
type Note struct {
	ID   int    `json:"id"`
	Text string `json:"text"`
}

// Service keeps the notes in creation order; ids start at 1.
type Service struct {
	mu    sync.Mutex
	notes []Note
}

var _ gen.ServerInterface = (*Service)(nil)

// New returns an empty service (the seam the tests serve).
func New() gen.ServerInterface { return &Service{notes: []Note{}} }

func write(w http.ResponseWriter, status int, v any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(v)
}

// ListNotes returns every note, in creation order.
func (s *Service) ListNotes(w http.ResponseWriter, r *http.Request) {
	s.mu.Lock()
	defer s.mu.Unlock()
	write(w, http.StatusOK, s.notes)
}

// CreateNote stores a note with the next id, or refuses an empty text (or one of only spaces)
// with a 400.
func (s *Service) CreateNote(w http.ResponseWriter, r *http.Request) {
	var in struct {
		Text string `json:"text"`
	}
	if err := json.NewDecoder(r.Body).Decode(&in); err != nil || strings.TrimSpace(in.Text) == "" {
		write(w, http.StatusBadRequest, map[string]string{"message": "the text of a note must not be empty"})
		return
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	n := Note{ID: len(s.notes), Text: in.Text}
	s.notes = append(s.notes, n)
	write(w, http.StatusCreated, n)
}
